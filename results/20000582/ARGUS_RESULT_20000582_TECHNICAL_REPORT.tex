\title[Moving-Window Engel Hilbert Cocycles with $\Omega(\sqrt{k})$ Gauge Distortion]
{Moving-Window Engel Hilbert Cocycles with $\Omega(\sqrt{k})$ Homogeneous Gauge Distortion}
\author{Argus Open Research Project}
\date{September 5, 2026}
\subjclass[2020]{Primary 51F30; Secondary 20F65, 46B85}
\begin{document}
\begin{abstract}
Toward the Hilbert-space Lang--Plaut bi-Lipschitz embedding problem for doubling metric spaces, we construct a family of moving-window injective Hilbert-space cocycles on finite balls of the Engel group that are uniformly doubling, admit an exact logarithmic gauge comparison, and incur distortion at least $c\sqrt{k}$ relative to any fixed homogeneous Engel gauge.
\end{abstract}
\maketitle
\section{Main Construction Theorem}
\begin{theorem}
On the three-step nilpotent Engel group $\mathcal{E}$, there exists a sequence of moving-difference-window maps $F_k : B_{\mathcal{E}}(e, R_k) \to \ell_2$ such that:
\begin{enumerate}[label=\textup{(\arabic*)}]
  \item The induced metric balls are uniformly doubling with doubling constant independent of $k$.
  \item Relative to the homogeneous Carnot--Carath\'eodory gauge $d_{\mathrm{CC}}$, the compression/distortion of $F_k$ on $B_{\mathcal{E}}(e, R_k)$ is at least $c\sqrt{k}$ for an explicit absolute constant $c>0$.
\end{enumerate}
\end{theorem}
