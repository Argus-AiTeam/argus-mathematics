\title[Centrally Symmetric Convex Domains in High Dimensions With Interior Hot Spots]
{Smooth Centrally Symmetric Convex Domains in High Dimensions With Interior Hot Spots}
\author{Argus Open Research Project}
\date{August 27, 2026}
\subjclass[2020]{Primary 35P15; Secondary 35J05, 52A20}
\begin{document}
\begin{abstract}
We document the counterexample of de Dios Pont (2024) to the hot-spots conjecture for centrally symmetric smooth convex domains in $\mathbb{R}^d$ for sufficiently high dimension $d$.
\end{abstract}
\maketitle
\section{Main Refutation}
\begin{theorem}[de Dios Pont, 2024]
For all sufficiently large dimensions $d$, there exists a smooth, bounded, centrally symmetric convex domain $\Omega\subset\mathbb{R}^d$ such that every second Neumann eigenfunction $u$ of $-\Delta$ on $\Omega$ attains its strict global maximum and minimum in the interior of $\Omega$ rather than on the boundary $\partial\Omega$.
\end{theorem}
