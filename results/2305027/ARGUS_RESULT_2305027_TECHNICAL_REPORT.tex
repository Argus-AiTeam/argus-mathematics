\title[Signed Cauchy--Green Flux Identity and Linear Residual Lower Bound for the Corona Problem]
{A Signed Cauchy--Green Flux Identity and Linear Residual Lower Bound on Planar Corona Corrections}
\author{Argus Open Research Project}

\begin{abstract}
For the corona problem on planar domains with $N$ radial slit/annular barriers, we prove an exact signed Cauchy--Green contour identity and show that for the derivative-coupled correction family $a_\nu = z^2 f_\nu'$, even when the radial trace cost is below $1/500{,}000$, the full literal $\bar{\partial}$-residual norm grows linearly at least as $N/400{,}000$ for all $N\ge 21$.
\end{abstract}
\maketitle
\section{Main Lower Bound}
\begin{theorem}
For every $N\ge 21$ and every cap-feasible density in the derivative-coupled corona correction family $a_\nu(z)=z^2 f_\nu'(z)$ on the $N$-chamber planar domain, the signed Cauchy--Green interior flux identity forces the complete literal residual to satisfy
\[
  \|R_N\|_{L^1} \;\ge\; \frac{N}{400{,}000},
\]
excluding uniform boundedness as $N\to\infty$ within that correction ansatz.
\end{theorem}
