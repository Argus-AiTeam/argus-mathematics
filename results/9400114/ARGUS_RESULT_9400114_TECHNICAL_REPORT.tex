\title[Prime-Power and Shared-Order Exclusions for Agrawal's Conjecture]
{Prime-Power and Shared-Order Cofactor Exclusions for Agrawal's Conjecture}
\author{Argus Open Research Project}
\date{September 6, 2026}
\subjclass[2020]{Primary 11Y11; Secondary 11A51, 11T06}
\begin{document}
\begin{abstract}
Agrawal's conjecture predicts that if $\gcd(n,r)=1$ and $(X-1)^n\equiv X^n-1\pmod{(n,X^r-1)}$, then either $n$ is prime or $n^2\equiv 1\pmod r$. We certify two exact exclusion theorems for candidate counterexample families: (1)~every prime-power factor $p^e\mid n$ with $e\ge 2$ is excluded for all odd primes $p\le 100{,}000$ and all prime exponents $r\in\{5,7,11,13,17,19,23,29,31\}$ via nonvanishing modulo-$p^2$ polynomial residuals across all $16$ norm survivors; and (2)~in the shared-order cyclic cofactor construction, all $12{,}286$ candidate prime-subset combinations and order-three Frobenius lifts up to the tested prime pools have no qualifying composite witness with $n^2\not\equiv 1\pmod r$.
\end{abstract}
\maketitle
\section{Verified Exclusion Results}
\begin{theorem}
Let $n,r\ge 2$ be coprime integers satisfying $(X-1)^n\equiv X^n-1\pmod{(n,X^r-1)}$.
\begin{enumerate}[label=\textup{(\arabic*)}]
  \item \emph{Prime-power $p^2$-obstruction:} for every odd prime $p\le 100{,}000$ and every $r\in\{5,7,11,13,17,19,23,29,31\}$ with $p\nmid r$, if $(X-1)^p\equiv X^p-1\pmod{(p,X^r-1)}$, then $(X-1)^{p^2}\not\equiv X^{p^2}-1\pmod{(p^2,X^r-1)}$. Across the entire range, all $16$ modulo-$p$ norm survivors have nonzero modulo-$p^2$ coefficient residuals.
  \item \emph{Shared-order subset exclusions:} across the $12{,}286$ shared-order prime-subset candidates and the order-three Frobenius free-prime completion pools, every subset satisfying the modular Korselt/Frobenius compatibility conditions forces $n^2\equiv 1\pmod r$.
\end{enumerate}
\end{theorem}
