\title[Universal Finite-Dimensionality of Repaired Bigelow Zipper Quotients]
{Universal Finite-Dimensionality of Repaired Bigelow Zipper Quotients for All Ranks $n\ge 4$}

\author{Argus Open Research Project}
\address{Argus Open Research Project}

\subjclass[2020]{Primary 20F36; Secondary 16S34, 57K10}

\begin{abstract}
We prove that under both natural well-typed repairs of Bigelow's zipper-algebra presentation, enlarged by the twist-compatibility relations and the rank-four vanishing relation $X_4=0$ at $u=q^3$, the universal presented quotient algebra $Q_n$ over $K=\mathbb{Q}(q)$ is finite-dimensional for every finite strand number $n\ge 4$, with explicit dimension bounds
\[
  (2n-1)!! + 1 \;\le\; \dim_K Q_n \;\le\; (2n-1)!! + \left(\frac{n!}{3}\right)^2.
\]
The lower bound follows from the surjection onto the specialized Birman--Murakami--Wenzl (BMW) algebra together with the nonzero rank-one unreduced Burau image of $X_3$ in the kernel. The upper bound is proved by an all-rank induction on the left and right principal modules $Q_n e$ and $e Q_n$ generated by the normalized idempotent $e=(q-1)^{-2}(q+1)^{-1}X_3$, combining the exact $169$-dimensional rank-four regular quotient ($\dim_K Q_4 e = \dim_K e Q_4 = 8 = 4!/3$) with algebraic tail annihilation $e(\sigma_i-1)(\sigma_i+q)=0$ ($i\ge 3$) and uniform two-sided Schreier-defect propagation.
\end{abstract}

\section{Main Theorem}

Let $K=\mathbb{Q}(q)$, and let $Q_n$ ($n\ge 4$) denote the universal $K$-algebra quotient of the braid group algebra $K B_n$ under either of the two well-typed repairs of Bigelow's zipper relations, together with the twist relations at $u=q^3$ and the vanishing of $X_4$. Set
\[
  s \;=\; (q-1)^2(q+1),\qquad e \;=\; s^{-1}X_3,\qquad P(T) \;=\; (T-1)(T+q).
\]
In $Q_n$, the quotient $Q_n / (X_3)$ is isomorphic to the specialized BMW algebra of dimension $(2n-1)!!$, so $\dim_K Q_n = (2n-1)!! + \dim_K J_n$, where $J_n = Q_n X_3 Q_n = Q_n e Q_n$ is the two-sided ideal generated by $e$.

\begin{theorem}[Universal Finite-Dimensionality and Dimension Bounds]\label{thm:main}
For every integer $n\ge 4$:
\begin{enumerate}[label=\textup{(\arabic*)}]
  \item The element $e$ is a nonzero idempotent in $Q_n$, and the principal left and right modules $Q_n e$ and $e Q_n$ satisfy
  \[
    \dim_K(Q_n e) \;\le\; \frac{n!}{3},\qquad \dim_K(e Q_n) \;\le\; \frac{n!}{3},
  \]
  with exact equality $\dim_K(Q_4 e)=\dim_K(e Q_4)=8$ at $n=4$.
  \item The multiplication map $(Q_n e)\otimes_{e Q_n e}(e Q_n)\twoheadrightarrow J_n = Q_n e Q_n$ satisfies $e Q_n e = K e$, and therefore
  \[
    1 \;\le\; \dim_K J_n \;\le\; \left(\frac{n!}{3}\right)^2,
    \qquad
    (2n-1)!! + 1 \;\le\; \dim_K Q_n \;\le\; (2n-1)!! + \left(\frac{n!}{3}\right)^2.
  \]
\end{enumerate}
\end{theorem}

\section{Exact Rank-Four Base and Algebraic Tail Annihilation}

In the exact $169$-dimensional characteristic-zero regular representation of $Q_4$ ($169 = 15 + 64 + 90$ prior to the twist specialization and $15+64=79$ with $\dim_K J_4=64$), left and right multiplication by $e$ satisfy $L_e^2=L_e$, $R_e^2=R_e$, and $\operatorname{tr}(L_e)=\operatorname{tr}(R_e)=8$. Hence $\dim_K(Q_4 e)=\dim_K(e Q_4)=8$, spanned by the Schreier bases
\[
  L_4 \;=\; \bigl\{t v : t\in\{1,\sigma_3,\sigma_2\sigma_3,\sigma_1\sigma_2\sigma_3\},\ v\in\{e,\sigma_1 e\}\bigr\},
  \qquad
  R_4 \;=\; \bigl\{v \bar{t} : v\in\{e,e\sigma_2\},\ \bar{t}\in\{1,\sigma_3,\sigma_3\sigma_2,\sigma_3\sigma_2\sigma_1\}\bigr\}.
\]
Moreover, the rank-four right action satisfies the quadratic annihilation $e P(\sigma_3) = 0$.

\begin{lemma}[All-Rank Quadratic Tail Annihilation]\label{lem:tail}
For every $n\ge 5$, one has $e P(\sigma_i) = 0$ for all $3\le i\le n-1$ and $P(\sigma_i) e = 0$ for all $4\le i\le n-1$.
\end{lemma}

\begin{proof}
Put $z=\sigma_3$ and $w=\sigma_4$. From the braid relation $zwz=wzw$, we have $wz P(w)=P(z)wz$ and $z^{-1}P(w)z = w P(z) w^{-1}$. Since $e$ is supported on strands $1,2,3$, $e$ commutes with $w=\sigma_4$. Multiplying by $e$ on the left and using $e P(z)=0$ yields $ez P(w)=0$ and $ez^{-1}P(w)=0$. Since $e P(z)=0$ expands to $ez^{-1} = q^{-1}ez + (q-1)q^{-1}e$, we obtain
\[
  0 \;=\; ez^{-1}P(w) \;=\; q^{-1}ez P(w) + (q-1)q^{-1}e P(w) \;=\; (q-1)q^{-1}e P(w),
\]
so $e P(\sigma_4)=0$. Inducting along the braid chain $\sigma_4,\dots,\sigma_{n-1}$ gives $e P(\sigma_i)=0$ for all $i\ge 3$, and the symmetric left argument starting from $P(\sigma_4)e=0$ gives $P(\sigma_i)e=0$ for all $i\ge 4$.
\end{proof}

\section{Uniform Schreier-Defect Propagation}

For $n\ge 4$ and $1\le j\le n$, set $d_j^{(n)}=\sigma_j\sigma_{j+1}\cdots\sigma_{n-1}$ and $\bar{d}_j^{(n)}=\sigma_{n-1}\cdots\sigma_j$ (with $d_n^{(n)}=\bar{d}_n^{(n)}=1$), and define recursively
\[
  L_3 = \operatorname{span}_K\{e,\sigma_1 e\},\quad L_n = \sum_{j=1}^n d_j^{(n)} L_{n-1},
  \qquad
  R_3 = \operatorname{span}_K\{e,e\sigma_2\},\quad R_n = \sum_{j=1}^n R_{n-1}\bar{d}_j^{(n)}.
\]
Clearly $\dim_K L_n \le n \dim_K L_{n-1} \le \cdots \le (n!/3)$ and $\dim_K R_n \le n!/3$. For any standard generator $\sigma_i$ ($1\le i<n$), the braid relations give
\[
  \sigma_i d_k^{(n)} \;=\;
  \begin{cases}
    d_k^{(n)}\sigma_i, & i < k-1, \\
    d_{k-1}^{(n)}, & i = k-1, \\
    (1-q)d_k^{(n)} + q d_{k+1}^{(n)} + P(\sigma_k)d_{k+1}^{(n)}, & i = k, \\
    d_k^{(n)}\sigma_{i-1}, & i > k.
  \end{cases}
\]
Thus $L_n$ is closed under left multiplication by $K B_n$ as soon as $P(\sigma_k)d_{k+1}^{(n)} L_{n-1}\subseteq L_n$ for $1\le k<n$. At $n=5$, the rank-four base identities and nonlocal syzygy
\[
  P(\sigma_4)(\sigma_3 e - e) \;=\; \frac{q-1}{q}\sigma_3\sigma_4 P(\sigma_3)e + (1-q)(\sigma_1+q)\sigma_2\sigma_3\sigma_4\sigma_2\sigma_3 e \;\in\; L_5
\]
together with the dual right identity close $L_5=Q_5 e$ and $R_5=e Q_5$. For $n\ge 6$, shifting the rank-four and rank-five defect relations by distant commutation and applying Lemma~\ref{lem:tail} propagates $P(\sigma_k)d_{k+1}^{(n)} L_{n-1}\subseteq L_n$ and $R_{n-1}\bar{d}_{k+1}^{(n)} P(\sigma_k)\subseteq R_n$ across all $1\le k<n$. Since $\sigma_i^{-1}\in\operatorname{span}_K\{1,\sigma_i,\sigma_i^2\}$ by the cubic Hecke/BMW relation, $L_n=Q_n e$ and $R_n=e Q_n$ hold for all $n\ge 4$, completing the proof of Theorem~\ref{thm:main}.\qed
