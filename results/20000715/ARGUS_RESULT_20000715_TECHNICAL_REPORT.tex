\title[Algebraic Differential Equations and Linear-ODE Obstructions for $\eta(\tau)^{24}$]
{A Third-Order Algebraic Differential Equation and All-Order $\mathbb{C}(q)$-Linear Obstruction for $\eta(\tau)^{24}$}
\author{Argus Open Research Project}
\date{September 7, 2026}
\subjclass[2020]{Primary 11F11; Secondary 12H05, 34M15}
\begin{document}
\begin{abstract}
In connection with AIM Problem~1.2 on differential equations for $\eta$-products, we record an explicit third-order nonlinear polynomial differential equation with rational coefficients satisfied by the discriminant modular form $y(q)=\Delta(\tau)=\eta(\tau)^{24}$ under $D=q\frac{d}{dq}$, and prove that $y(q)$ satisfies no nonzero finite-order homogeneous linear differential equation over $\mathbb{C}(q)$.
\end{abstract}
\maketitle
\section{Main Theorem}
Let $q=e^{2\pi i\tau}$ ($\operatorname{Im}\tau>0$), $D=q\frac{d}{dq}=\frac{1}{2\pi i}\frac{d}{d\tau}$, and $y(q)=\eta(\tau)^{24}=q\prod_{n=1}^\infty(1-q^n)^{24}$.
\begin{theorem}
The function $y(q)=\eta(\tau)^{24}$ satisfies:
\begin{enumerate}[label=\textup{(\arabic*)}]
  \item \emph{Third-order nonlinear algebraic ODE:} eliminating the Eisenstein series $E_2,E_4,E_6$ from Ramanujan's differential system $DE_2=(E_2^2-E_4)/12$, $DE_4=(E_2E_4-E_6)/3$, $DE_6=(E_2E_6-E_4^2)/2$, together with $Dy = E_2 y$ and $1728y = E_4^3 - E_6^2$, yields the exact homogeneous degree-12 polynomial identity in $(y, Dy, D^2y, D^3y)$:
  \[
    \bigl(24\,y^2 D^3y - 90\,y\,Dy\,D^2y + 65\,(Dy)^3\bigr)^2
    - \bigl(12\,y\,D^2y - 13\,(Dy)^2\bigr)^3
    - 1728\,y^6 \;=\; 0.
  \]
  \item \emph{All-order homogeneous $\mathbb{C}(q)$-linear obstruction:} for every integer $m\ge 0$ and every tuple of rational functions $(a_0(q),\dots,a_m(q))\in\mathbb{C}(q)^{m+1}\setminus\{0\}$, one has $\sum_{j=0}^m a_j(q) D^j y(q) \not\equiv 0$.
\end{enumerate}
\end{theorem}
\begin{proof}[Proof of (2)]
By clearing denominators, assume $a_j(q)\in\mathbb{C}[q]$ are not all zero. Select a root of unity $\zeta=e^{2\pi i c/d}$ at which not all $a_j(\zeta)$ vanish (or after dividing out $(q-\zeta)^r$). Under the modular transformation $\tau = c/d + i t$ as $t\to 0^+$, $\eta(\tau)^{24} = (d t)^{-12} e^{-2\pi/(d^2 t)}(1 + O(e^{-2\pi/(d^2 t)}))$ vanishes to infinite order along the radial ray $q=e^{-2\pi t}\zeta\to\zeta$, so $\lim_{t\to 0^+} (q-\zeta)^{-k} D^j y(q) = 0$ for all $j,k\ge 0$, contradicting the leading nonvanishing Taylor term of $\sum_{j=0}^m a_j(q) D^j y(q)$ at $q=\zeta$.
\end{proof}
