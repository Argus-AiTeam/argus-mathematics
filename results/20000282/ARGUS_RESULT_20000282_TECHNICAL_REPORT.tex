\title[A Reduced Non-Equidimensional Counterexample to Hilbert--Kunz Upper Semicontinuity]
{Failure of Upper Semicontinuity of Hilbert--Kunz Multiplicity on a Reduced Non-Equidimensional Ring}
\author{Argus Open Research Project}
\date{August 27, 2026}
\subjclass[2020]{Primary 13A35; Secondary 13D40}
\begin{document}
\begin{abstract}
Problem~1.5 of the AIM problem list on test ideals and multiplier ideals asks to investigate the upper semicontinuity of $\mathfrak{p}\mapsto e_{\mathrm{HK}}(R_{\mathfrak{p}})$ on $\operatorname{Spec} R$, without imposing a local-equidimensionality hypothesis. Over $k=\mathbb{F}_p$, we prove that the reduced finite-type ring $R=k[a,b,c,d]/(ab,acd)$ has prime ideals $\mathfrak{p}=(b,c,d)\subsetneq\mathfrak{m}=(a,b,c,d)$ satisfying $e_{\mathrm{HK}}(R_{\mathfrak{p}})=2 > 1 = e_{\mathrm{HK}}(R_{\mathfrak{m}})$, so the sublevel set $\{\mathfrak{q}\in\operatorname{Spec} R : e_{\mathrm{HK}}(R_{\mathfrak{q}})<3/2\}$ is not Zariski open.
\end{abstract}
\maketitle
\section{Exact Monomial-Length Calculation and Proof}
Let $k=\mathbb{F}_p$ ($p>0$ prime), $A=k[a,b,c,d]$, $I=(ab,acd)$, and $R=A/I$.
\begin{theorem}
In $R=k[a,b,c,d]/(ab,acd)$:
\begin{enumerate}[label=\textup{(\arabic*)}]
  \item $I = (a)\cap(b,c)\cap(b,d)$ is an irredundant intersection of prime ideals of dimensions $3,2,2$, so $R$ is reduced, excellent, and $F$-finite, but not locally equidimensional at $\mathfrak{m}=(a,b,c,d)$.
  \item At the prime $\mathfrak{p}=(b,c,d)\subsetneq\mathfrak{m}$, $a\notin\mathfrak{p}$ is a unit, so $R_{\mathfrak{p}}\cong k(a)[c,d]_{(c,d)}/(cd)$ has dimension~$1$, standard monomial basis $\{1,c,\dots,c^{q-1},d,\dots,d^{q-1}\}$ modulo $\mathfrak{p}^{[q]}$ of length $2q-1$ for every $q=p^e$, and Hilbert--Kunz multiplicity
  \[
    e_{\mathrm{HK}}(R_{\mathfrak{p}}) \;=\; \lim_{q\to\infty}\frac{2q-1}{q^1} \;=\; 2.
  \]
  \item At the maximal ideal $\mathfrak{m}=(a,b,c,d)$, $\dim R_{\mathfrak{m}}=3$, and counting standard monomials $a^i b^j c^r d^s$ ($0\le i,j,r,s<q$) avoiding $ab$ and $acd$ gives $q^3$ monomials with $i=0$ and $(q-1)(2q-1)$ monomials with $1\le i<q$ and $j=0$, for a total length of $q^3+2q^2-3q+1$ and Hilbert--Kunz multiplicity
  \[
    e_{\mathrm{HK}}(R_{\mathfrak{m}}) \;=\; \lim_{q\to\infty}\frac{q^3+2q^2-3q+1}{q^3} \;=\; 1.
  \]
\end{enumerate}
Since $\mathfrak{p}\subsetneq\mathfrak{m}$ and $e_{\mathrm{HK}}(R_{\mathfrak{p}})=2 > 1 = e_{\mathrm{HK}}(R_{\mathfrak{m}})$, the function $\mathfrak{q}\mapsto e_{\mathrm{HK}}(R_{\mathfrak{q}})$ is not upper semicontinuous on $\operatorname{Spec} R$.
\end{theorem}
