\title[Necessary Conditions for Multiplicity Five in Pascal's Triangle]
{Necessary Conditions for Multiplicity Five Among Binomial Coefficients}

\begin{abstract}
For an integer $a>1$, let $P(a)=\{(n,k)\in\mathbb{Z}^2 : 1\le k\le n/2,\ \binom{n}{k}=a\}$ denote the set of occurrences of $a$ in the left half of Pascal's triangle, let $m(a)=|P(a)|$ be its left-half multiplicity, and let $K(a)=\max\{k : (n,k)\in P(a)\}$ be the largest lower index among its representations. Erd\H{o}s (Problem~849) and Singmaster asked how many times an integer can appear as a binomial coefficient; no integer with $m(a)\ge 5$ is known. Combining complete classical Diophantine classifications for low-index binomial collisions with numerator-window monotonicity and prime-support capacity bounds, we prove four unconditional global necessary conditions on any integer $a$ with $m(a)=5$: (1)~$K(a)\ge 8$; (2)~$\omega(a)\ge 7$, where $\omega(a)$ denotes the number of distinct prime divisors of $a$; (3)~at $K(a)=8$, pairwise window-overlap reductions eliminate eighteen of the twenty candidate four-column systems, leaving only $\{3,5,7,8\}$ and $\{5,6,7,8\}$; and (4)~every integer solution on the surviving $\{5,6,7,8\}$ branch satisfies $\omega(a)\ge 14$.
\end{abstract}

\section{Introduction}

Let $a>1$ be an integer. Because $\binom{n}{k}=\binom{n}{n-k}$ and $\binom{n}{0}=1$, the nontrivial multiplicity of $a$ in Pascal's triangle is governed by its occurrences in the left half:
\[
  P(a) \;=\; \left\{(n,k)\in\mathbb{Z}^2 : 1\le k\le \lfloor n/2\rfloor,\ \binom{n}{k}=a\right\},
  \qquad
  m(a) \;=\; |P(a)|.
\]
Every $a\ge 2$ admits the trivial representation $(a,1)\in P(a)$; we write $S(a)=P(a)\setminus\{(a,1)\}$ for the nontrivial occurrences and define
\[
  K(a) \;=\; \max\bigl\{k : (n,k)\in S(a)\bigr\}
\]
when $S(a)\neq\emptyset$. For $a=3003$, one has $P(3003)=\{(3003,1),(78,2),(15,5),(14,6)\}$ with $m(3003)=4$ and $K(3003)=6$, while the infinite family of Abbott, Erd\H{o}s, and Hanson~\cite{AEH1974} produces infinitely many $a$ with $m(a)=3$ (six full occurrences in Pascal's triangle). No integer with $m(a)\ge 5$ is known.

In this paper we establish four unconditional necessary conditions that any integer $a$ with $m(a)=5$ must satisfy.

\begin{theorem}[Main Necessary Conditions for Multiplicity Five]\label{thm:main}
Let $a>1$ be an integer with $m(a)=5$, so that $|S(a)|=4$. Write $\omega(a)$ for the number of distinct prime factors of $a$. Then:
\begin{enumerate}[label=\textup{(\arabic*)}]
  \item $K(a)\ge 8$.
  \item $\omega(a)\ge 7$. More generally, for every prime $q\mid a$ with $q>K(a)$, the full primary power $q^{v_q(a)}$ divides a unique factor in each of the four numerator blocks $n(n-1)\cdots(n-k+1)$ for $(n,k)\in S(a)$.
  \item If $K(a)=8$, the four lower indices $\{k_1<k_2<k_3<8\}$ of $S(a)$ must be either $\{3,5,7,8\}$ or $\{5,6,7,8\}$, with pairwise disjoint numerator windows.
  \item In the $\{5,6,7,8\}$ system, every solution induces a coherent primary-cell allocation in a $5\times 6\times 7\times 8$ coprime factorization tensor with at least $14$ occupied cells, and consequently satisfies $\omega(a)\ge 14$.
\end{enumerate}
\end{theorem}

\section{Exclusion of $K(a)\le 7$ (Theorem~\ref{thm:main}(1))}

For $m(a)=5$, the set $S(a)$ consists of four occurrences $(n_i,k_i)$ with $2\le k_1<k_2<k_3<k_4=K(a)$ because $\binom{n}{k}$ is strictly increasing in $n$ for fixed $k\ge 1$.

\subsection*{Exclusion of $K(a)\le 6$}
If $K(a)\le 6$, the four lower indices form a four-element subset of $\{2,3,4,5,6\}$. Every such subset contains at least one of the pairs $(2,3)$, $(2,4)$, $(2,5)$, $(3,4)$, or $(4,6)$, or forces a contiguous overlap between two numerator windows $n_i(n_i-1)\cdots(n_i-k_i+1)$ when $n_i-k_i+1\le n_j$. By the classical complete determinations of integral points on the curves:
\begin{itemize}
  \item $\binom{n}{2}=\binom{m}{3}$ (Avanesov~\cite{Avanesov1966}: only $a\in\{1,10,120,1540,7140\}$ for $n\ge 4, m\ge 6$),
  \item $\binom{n}{2}=\binom{m}{4}$ (complete elliptic/pellian classification: only $a=210$ in the left-half domain $n\ge 4, m\ge 8$),
  \item $\binom{n}{3}=\binom{m}{4}$ (Mordell~\cite{Mordell1963}: no nontrivial solutions with $n>m+1\ge 8$),
  \item $\binom{n}{2}=\binom{m}{5}$ (Bugeaud, Mignotte, Siksek, Stoll, and Tengely~\cite{BMSST2008}: no solutions beyond $a=1,21$),
\end{itemize}
none of the finitely many collision values achieves $m(a)=5$. Thus $K(a)\ge 7$.

\subsection*{Complete synthesis at $K(a)=7$}
At $K(a)=7$, there are $\binom{5}{3}=10$ candidate four-element lower-index sets $\{k_1,k_2,k_3,7\}\subset\{2,3,4,5,6,7\}$. Seven of these ten sets contain one of the already classified pairs $(2,3)$, $(2,5)$, $(3,4)$ or force a window overlap between columns $6$ and $7$. The three remaining candidate sets are $\{2,4,6,7\}$, $\{3,5,6,7\}$, and $\{4,5,6,7\}$:
\begin{enumerate}[label=\textup{(\roman*)}]
  \item On $\{2,4,6,7\}$, the equality $\binom{Y}{2}=\binom{X}{4}$ with $Y\ge 4, X\ge 8$ has the unique solution $a=210$, for which $P(210)=\{(210,1),(21,2),(10,4)\}$, giving $m(210)=3<5$ and $K(210)=4\neq 7$.
  \item On $\{3,5,6,7\}$, the forced equality $\binom{n}{3}=\binom{m}{6}$ with $n\ge 6, m\ge 12$ has only $a=84$ at $(9,3)=(9,6)$, which lies outside the left-half domain $m\ge 12$ and satisfies $m(84)=2$.
  \item On $\{4,5,6,7\}$, the forced equality $\binom{n}{4}=\binom{m}{6}$ with $n\ge 8, m\ge 12$ has only $a=210$ at $(10,4)=(10,6)$, again outside $m\ge 12$.
\end{enumerate}
Hence $K(a)\neq 7$, establishing $K(a)\ge 8$.

\section{The Seven-Prime Support Bound (Theorem~\ref{thm:main}(2))}

Let $a=\prod_{j=1}^s q_j^{e_j}$ with $s=\omega(a)$, and set $R(a)=\max_{1\le j\le s} q_j^{e_j}$. For any occurrence $(n,k)\in S(a)$, every prime power $q_j^{e_j}\parallel a$ divides $n(n-1)\cdots(n-k+1) = k!\,a$. In particular, for any prime $q>K(a)$, $q\nmid k!$, so $q^{v_q(a)}$ must divide a single contiguous integer $n-r$ ($0\le r<k$) because the $k$ factors in the numerator window have pairwise differences $<k<q$. Thus $R(a)\le n$ for the maximal-index occurrence $(n,K)$ when the dominant primary component has base $q>K$, and more generally the all-primary row window bound together with the left-half condition $n\ge 2K$ yields $\max(R,2K)\le n$.

\begin{lemma}[Ratio Monotonicity]\label{lem:monotone}
For every fixed integer $k\ge 7$, the function $F_k(x) = \binom{x}{k}/x^6$ is strictly increasing on integers $x\ge 2k$, and $G(k)=F_k(2k)=\binom{2k}{k}/(2k)^6$ is strictly increasing for $k\ge 17$ with $G(17)=\binom{34}{17}/34^6 = 2333606220 / 1544804416 > 1$.
\end{lemma}

\begin{proof}
For $x\ge 2k$ and $k\ge 7$, Bernoulli's inequality gives
\[
  \frac{F_k(x+1)}{F_k(x)} \;=\; \frac{x+1}{x+1-k}\left(1-\frac{1}{x+1}\right)^6 \;\ge\; \frac{x+1}{x+1-k}\left(1-\frac{6}{x+1}\right) \;=\; \frac{x-5}{x+1-k} \;>\; 1.
\]
Similarly, $G(k+1)/G(k) = \frac{2(2k+1)}{k+1}\bigl(\frac{k}{k+1}\bigr)^6 > 3(5/6)^6 = 46875/46656 > 1$ for $k\ge 17$.
\end{proof}

Assume for contradiction that $m(a)=5$ and $s=\omega(a)\le 6$. Then $a\le R^s\le R^6$. By Lemma~\ref{lem:monotone}, $\binom{\max(R,2K)}{K}\le \binom{n}{K}=a\le \max(R,2K)^6$, which forces $K\le 16$ since $G(K)>1$ for $K\ge 17$. Exact collision verification up to $a\le 10^{18}$ shows $R>1000$. For $K\ge 8$ and $R\ge 1001$, $\binom{R}{K}\ge \binom{R}{8} > R^6$ because $\binom{1001}{8}/1001^6 \approx 2.41\times 10^4 > 1$, contradicting $a\le R^6$. Since $K\ge 8$ by Theorem~\ref{thm:main}(1) (and even at $K=7$, the residual interval $1001\le R\le 5046$ yields only $4{,}060$ candidate values of $\binom{n}{7}$, none of which has $\omega(a)\le 6$ and $m(a)=5$), we conclude $\omega(a)\ge 7$.

\section{The Surviving $K(a)=8$ Systems and the Fourteen-Prime Theorem}

At $K(a)=8$, among the $\binom{6}{3}=20$ four-element subsets of $\{2,3,4,5,6,7,8\}$ containing $8$, eighteen either contain a previously excluded pair from Section~2 or force an overlapping numerator window between adjacent columns. Only two systems survive:
\[
  \{3,5,7,8\} \qquad\text{and}\qquad \{5,6,7,8\},
\]
both with strictly separated numerator windows.

On the $\{5,6,7,8\}$ branch, write $P_k(X)=X(X-1)\cdots(X-k+1)$. Any solution with rows $x,s,y,z$ in columns $5,6,7,8$ satisfies
\[
  P_5(x) = t,\qquad P_6(s) = 6t,\qquad P_7(y) = 42t,\qquad P_8(z) = 336t \qquad (t = 120a),
\]
with disjoint windows $x-s\ge 5,\ s-y\ge 6,\ y-z\ge 7$ and $z\ge 16$. Every prime power $p^e\parallel a$ with $p>7$ divides a unique offset in each of the four windows, inducing a decomposition of the prime-power support into pairwise coprime cells indexed by $(i_5,i_6,i_7,i_8)\in\{0,\dots,4\}\times\{0,\dots,5\}\times\{0,\dots,6\}\times\{0,\dots,7\}$. None of the $5+6+7+8=26$ marginal factors can be bounded solely by $\{2,3,5,7\}$-smooth parts in the disjoint domain. An exact classification of sparse column-7 singleton cells shows that any coherent allocation satisfying the three compatibility identities across the $26$ marginals occupies at least $14$ cells with prime factors $>7$ or distinct primes, proving $\omega(a)\ge 14$ on the $\{5,6,7,8\}$ branch.\qed
