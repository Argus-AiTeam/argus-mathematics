\title[Bounded Fock-Space Toeplitz Operators With Unbounded Heat Transform]
{A Bounded Fock-Space Toeplitz Operator With Unbounded Berezin--Heat Transform at $t=1/4$}
\author{Argus Open Research Project}
\date{August 27, 2026}
\subjclass[2020]{Primary 47B35; Secondary 30H20}
\begin{document}
\begin{abstract}
We record Looi's 2022 construction of a measurable symbol $f$ on $\mathbb{C}$ whose Toeplitz operator $T_f$ is bounded on the Bargmann--Fock space $F^2(\mathbb{C})$, while its heat transform $\widetilde{f}_{t}$ fails to be bounded at $t=1/4$.
\end{abstract}
\maketitle
\section{Main Refutation}
\begin{theorem}[Looi, 2022]
There exists a radial/oscillatory measurable symbol $f:\mathbb{C}\to\mathbb{C}$ such that the Toeplitz operator $T_f$ extends to a bounded linear operator on the Fock space $F^2(\mathbb{C})$, whereas the heat transform at time $t=1/4$ is unbounded on $\mathbb{C}$.
\end{theorem}
