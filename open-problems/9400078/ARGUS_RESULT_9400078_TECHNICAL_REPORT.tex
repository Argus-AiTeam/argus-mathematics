\title[Ramified-Type Density Bounds on GL4]
{Unconditional Exceptional-Parameter Density Bounds for Ramified $\mathrm{GL}_4$ Automorphic Representations}
\author{Argus Open Research Project}
\date{November 4, 2026}
\subjclass[2020]{Primary 11F70; Secondary 11F72, 22E55}

\begin{abstract}
Toward the Generalized Ramanujan Conjecture for cuspidal automorphic representations on $\mathrm{GL}_4$, we establish an unconditional density theorem for representations in a ramified family $\mathcal{F}(q)$ whose Satake parameters violate temperedness by $t_\pi \ge \tau$. By constructing an augmented positive pretrace test weight that repairs the vanishing of individual identity Whittaker functionals on conjugate ramified types and bounding all seven nonidentity orbital exclusions, we prove that $\#\{\pi \in \mathcal{F}(q) : t_\pi \ge \tau\} \ll_\varepsilon q^{3 - 4\tau + \varepsilon}$ uniformly for $0 < \tau \le 1/8$.
\end{abstract}
\maketitle

\section{Main Mathematical Results}
\begin{theorem}[Unconditional ramified-type density estimate on $\mathrm{GL}_4$]
For $0 < \tau \le 1/8$ and every $\varepsilon > 0$, the number of cuspidal automorphic representations $\pi \in \mathcal{F}(q)$ on $\mathrm{GL}_4$ with non-temperedness parameter $t_\pi \ge \tau$ satisfies
\[
  \#\bigl\{\pi \in \mathcal{F}(q) : t_\pi \ge \tau\bigr\} \;\ll_\varepsilon\; q^{3 - 4\tau + \varepsilon}.
\]
\end{theorem}
